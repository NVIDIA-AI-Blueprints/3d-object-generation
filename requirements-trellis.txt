# Mirrors the "--basic" package set from microsoft/TRELLIS's own setup.sh,
# minus easydict and kaolin: this project's trellis_patch/ files remove the
# only usages of both (see setup_trellis.bat / setup_trellis.sh), replacing
# kaolin's mesh extraction with a vendored MIT-licensed FlexiCubes and
# diff-gaussian-rasterization (INRIA license) with the Apache-2.0 gsplat.
#
# transformers is intentionally omitted here: requirements.txt already pins
# a newer version and installing both would fight over the version.
#
# NOTE: the CUDA extensions (vox2seq, nvdiffrast, diffoctreerast, gsplat) are
# NOT listed here - they require --no-build-isolation to see the already
# installed torch/CUDA toolchain, which a plain `pip install -r` can't apply
# selectively. They are installed as separate steps by install_linux.sh
# (matching how install_dependencies.py installs them on Windows).

pillow
imageio
imageio-ffmpeg
tqdm
opencv-python-headless
scipy
ninja
rembg
onnxruntime
trimesh
open3d
xatlas
pyvista
pymeshfix
igraph

# Sparse tensor backend TRELLIS's core (non-patched) code uses (config.py: SPCONV_ALGO)
spconv-cu120

# TRELLIS's sparse-attention module (trellis/modules/sparse/) hardcodes its
# attention backend to flash_attn, which is never installed by this project
# on either platform. xformers is the other backend it supports and has
# prebuilt wheels (flash_attn only ships source, and is a very slow build).
# Pinned to the exact version built against torch==2.7.0 (requirements-torch.txt) -
# `pip install xformers` with no pin resolves to a version requiring a much
# newer torch and will silently upgrade/break it.
xformers==0.0.30
